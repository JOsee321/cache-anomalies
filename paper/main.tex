\begin{abstract}
SIEVE (Zhang et al., NSDI '24) achieves strong miss-ratio reductions over LRU and CLOCK via a single FIFO queue, one \texttt{visited} bit per object, and a roving eviction \texttt{hand} that preserves spatial locality near \texttt{head}. We identify and prove a structural pathology unique to SIEVE's decoupled insertion/eviction geometry: \emph{hand starvation (zombie pinning)}. We prove that for every $3 \le k < C$, an explicit trace of prefix length $2C-2$ and a 3-key cycle yields a \textbf{0\% steady-state miss rate} in $\mathrm{SIEVE}(k)$ versus a \textbf{100\% steady-state miss rate} in $\mathrm{SIEVE}(C)$, pinning $C-2$ unvisited zombie keys indefinitely—even under $1\%$ disjoint Zipf noise or $25\%$ random hits to the pinned zombies. We prove a sharp phase transition at $k=2$, establish a structural separation theorem showing that FIFO, CLOCK, 2-bit CLOCK, SecondChance, and S3-FIFO are all immune to $0\%$ vs.\ $>0\%$ starvation (while mapping periodic $2.27\times$ Belady spikes in S3-FIFO at $C=10m$), and evaluate two $O(1)$ zero-per-node-overhead mitigations (\texttt{SIEVE-BW} and \texttt{SIEVE-TP}) across a symmetric Pareto frontier $\{C, 2C, 4C, 8C, 16C\}$.
\end{abstract}

\section{Introduction}
FIFO-based eviction policies such as S3-FIFO (SOSP '23) and SIEVE (NSDI '24) eliminate list relinking on cache hits while outperforming LRU on skewed workloads. Because these policies lack the stack inclusion property, they remain susceptible to Belady's anomaly. While Fornai and Iv\'anyi (2010) proved unbounded Belady ratios for classic FIFO when both capacities miss, and SR-Gadgets (2026) targets repeated-scan cliffs, the stationary-\texttt{hand} starvation pathology of SIEVE has not been formally characterized.

\section{SIEVE Semantics and Edge-Case Sensitivity}
SIEVE orders nodes from \texttt{head} (newest) to \texttt{tail} (oldest) with a roving pointer \texttt{hand} (initially \texttt{None}). Direct line-by-line audit against upstream \texttt{1a1a11a/libCacheSim} (commit \texttt{4b66d97}, \texttt{Sieve.c:126-245} and \texttt{cache.c:321-326}) confirms 1:1 pointer conformance: on hit, \texttt{node.visited = True} without moving the node; on miss when full, starting at $o = \texttt{hand}$ (or \texttt{tail} if \texttt{None}), SIEVE clears \texttt{o.visited = False} and steps $o \leftarrow o.\texttt{prev}$ (wrapping to \texttt{tail}) until finding \texttt{o.visited == False}; it then sets $\texttt{hand} \leftarrow o.\texttt{prev}$, unlinks $o$, and subsequently inserts the new item at \texttt{head} with \texttt{visited = False}. Meanwhile, for S3-FIFO, our comparative baseline models paper-prose \texttt{move-to-main-threshold=1}; verifying exact \texttt{libCacheSim} \texttt{S3FIFO.c} / \texttt{S3FIFOv0.c} (\texttt{move-to-main-threshold=2}, total-capacity eviction, \texttt{int64\_t(0.90*C)} ghost sizing) in \texttt{cmd/conformance\_check} confirms identical Zombie Pinning immunity and multiple-of-10 boundary Belady spikes ($1.2833\times$ at $(19,20)$, $1.4999\times$ at $(29,30)$, $1.7044\times$ at $(39,40)$).

Inserting the new node at \texttt{head} \emph{before} updating $\texttt{hand} \leftarrow o.\texttt{prev}$ alters behavior whenever $o = \texttt{head}$ (setting \texttt{hand} to the newly inserted node instead of \texttt{None}). This diverges from the official specification on $700/700$ random traces and invalidates all 6 initial empirical witnesses ($k=3..8$), yet Zombie Pinning persists ($378/378$ pairs) because steady-state evictions always target index 1 ($o \ne \texttt{head}$).

\section{Formal Theorems}

\begin{theorem}[Zombie Pinning / Unbounded Anomaly in SIEVE]
For any integers $3 \le k < C$, let $z_1, \dots, z_{C-2}, A, B, D$ be $C+1$ distinct keys, with
\[
\mathrm{prefix} = (z_1, \dots, z_{C-2}, A, B, z_1, \dots, z_{C-2}), \qquad \mathrm{cycle} = (D, A, B).
\]
Under repeated execution of $\mathrm{cycle}$, $\mathrm{SIEVE}(C)$ locks into a $100\%$ steady-state miss rate while permanently pinning $z_1, \dots, z_{C-2}$, whereas $\mathrm{SIEVE}(k)$ converges to a $0\%$ steady-state miss rate.
\end{theorem}

\begin{proof}
Because $\mathrm{prefix}$ has $C$ distinct keys, $\mathrm{SIEVE}(C)$ experiences zero evictions during $\mathrm{prefix}$; after the second pass over $z_1\dots z_{C-2}$, its state is $[B(F), A(F), z_{C-2}(T), \dots, z_1(T)]$ with $\texttt{hand} = \texttt{None}$. On the first access $D$ of Cycle~0, \texttt{hand} scans from \texttt{tail} ($z_1$), resets all $z_i.\texttt{visited} = F$, stops at $o = A(F)$ (index 1 $\ne \texttt{head}$), sets $\texttt{hand} = B$, and inserts $D$ at \texttt{head}, yielding $[D(F), B(F), z_{C-2}(F), \dots, z_1(F)]$ with $\texttt{hand} = B$. Requests $A$ and $B$ symmetrically miss and evict index 1 ($B$ then $D$), returning to $[B(F), A(F), z_{C-2}(F), \dots, z_1(F)]$ with $\texttt{hand} = A$ (index 1). By induction, every access in every subsequent cycle misses at index 1 and leaves $z_1,\dots,z_{C-2}$ untouched forever. In $\mathrm{SIEVE}(k)$ ($k \ge 3$), the cycle touches only 3 keys $\{D,A,B\}$, so once all three co-reside in $\mathrm{SIEVE}(k)$, misses cease permanently (verified from cycle index 1 across all $378$ pairs $3 \le k < C \le 30$ and up to $(500,1000)$).
\end{proof}

\begin{theorem}[Sharp Phase Transition at $k=2$]
For $\mathrm{SIEVE}(2)$ and any $C \ge 3$, no reachable steady-state cycle has $0\%$ miss rate at size $2$ and $>0\%$ miss rate at size $C$.
\end{theorem}

\begin{proof}
A $0\%$-miss cycle $W$ in $\mathrm{SIEVE}(2)$ accesses $|D(W)| \le 2$ distinct keys. If $D(W) = \{A,B\}$, once $A$ resides in $\mathrm{SIEVE}(C)$ ($C \ge 3$), a miss on $B$ places $B$ at \texttt{head} and can evict $A$ only if \texttt{hand} reaches \texttt{head}, resetting $\texttt{hand}=\texttt{None}$. The next miss ($A$) wraps to \texttt{tail} ($\ne B$ since $C \ge 3$), retaining both $A$ and $B$ permanently.
\end{proof}

\begin{theorem}[Structural Separation]
None of FIFO, CLOCK, CLOCK2, SecondChance, or S3-FIFO can exhibit $0\%$ steady-state misses at size $k < C$ and $>0\%$ at size $C$.
\end{theorem}

\begin{proof}
Any $0\%$-miss cycle $W$ at size $k$ satisfies $|D(W)| \le k < C$. Unaccessed non-$D(W)$ keys are evicted within $\le C$ misses by FIFO, $\le 2C$ by CLOCK/SecondChance, and $\le 3C$ by CLOCK2. In S3-FIFO ($S = \max(1,\lfloor 0.1 C\rfloor)$, $M = C-S$, Ghost $G = M$), $S + M = C > |D(W)|$ guarantees that once prefix keys drain from $G$, no key of $D(W)$ evicted from $S$ can be displaced from $G$ before re-access, promoting all of $D(W)$ into $S \cup M$.
\end{proof}

\begin{proposition}[Bounded Recovery]
Both $\mathrm{SIEVE\text{-}BW}(C,K)$ (forcing $\texttt{hand}=\texttt{None}$ after $K$ unwrapped evictions) and $\mathrm{SIEVE\text{-}TP}(C,K)$ (probing \texttt{tail} after $K$ unwrapped evictions, directly evicting if unvisited or moving \texttt{tail} to \texttt{head} with \texttt{visited=F} if visited) break pure Zombie Pinning on eviction $K+1$ ($\lceil (K+1)/3\rceil$ cycles), and $\mathrm{SIEVE\text{-}TP}$ guarantees progress even under continuous \texttt{visited} refreshes on pinned keys.
\end{proposition}
